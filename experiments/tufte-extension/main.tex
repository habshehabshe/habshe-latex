\newcommand{\imageplaceholder}[2][\linewidth]{%
  \noindent

% -------------------------------------------------------------------
% Caption helpers
% Suppress prefix ("Figure 1: ") without breaking Tufte side-captions
% -------------------------------------------------------------------
\makeatletter
\newcommand{\suppresscaptionlabel}{%
  \def\fnum@figure{}%
  \def\fnum@table{}%
}
\makeatother

% -------------------------------------------------------------------
% Layout Environments: Notes (Full, Body, Margin)
% Unnumbered, non-floating, smaller academic typography
% -------------------------------------------------------------------
\NewDocumentEnvironment{Fnote}{ +b }{%
  \begin{fullwidth}%
    \smallskip
    \begingroup\footnotesize #1\par\endgroup
    \smallskip
  \end{fullwidth}%
}

\NewDocumentEnvironment{Bnote}{ +b }{%
  \par\smallskip
  \begingroup\footnotesize #1\par\endgroup
  \smallskip
}

\NewDocumentEnvironment{Mnote}{ O{0pt} +b }{%
  \marginnote[#1]{\footnotesize #2}%
}

% -------------------------------------------------------------------
% Generic Attribution & Reference Commands (Usable anywhere)
% -------------------------------------------------------------------
\makeatletter
\newcommand{\attribution}[1]{%
  \par\smallskip
  \begingroup
  \normalfont\footnotesize
  \noindent\textemdash\space#1\par
  \endgroup
}

\newcommand{\reference}[1]{%
  \par\vspace{2pt}% Secondary caption spacing
  \begingroup
  \normalfont\footnotesize
  \noindent
  \ifcsname b@#1\endcsname
    \bibentry{#1}%
  \else
    \nocite{#1}% Register with BibTeX
    #1%
  \fi
  \par
  \endgroup
}
\makeatother
